$env:PYTHONIOENCODING = "utf-8"
$env:PYTHONUTF8 = "1"

# Įrankių versijos
python -m pip show radon complexipy pydeps ruff pylint

# Ciklomatinis sudėtingumas (CC)
python -m radon cc app -s -a

# Maintainability Index (MI)
python -m radon mi app -s

# Kodo dydis (LOC, SLOC, LLOC)
python -m radon raw app -s

# Kognityvinis sudėtingumas
complexipy app

# Modulių priklausomybės (Fan-in / Fan-out analizei)
python -m pydeps app --show-deps --no-show --no-output

# Ruff statinė analizė
python -m ruff check app --output-format full

# Pylint statinė analizė
python -m pylint app --output-format=text
